# Bronze Era COA culture filters for imported Total War style symbols.
bronze_tw_coa_def_gaulish_trigger = {
	scope:actor ?= {
		exists = culture
		culture = { has_culture_group = culture_group:gaulish_bronze_group }
	}
}

bronze_tw_coa_def_baltic_finnic_trigger = {

		exists = culture

		}
	}
}

bronze_tw_coa_def_punic_trigger = {
	scope:actor ?= {
		OR = {
			AND = { exists = religion religion = religion:punic }
			AND = { exists = culture culture = culture:phoenician_punic }
		}
	}
}
