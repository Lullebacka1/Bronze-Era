# ============================================================================
# CITY PRESTIGE - BUILDING POINTS
# ============================================================================
# Root scope: location
#
# This is the only file that must be edited to make an urban building award
# City Prestige points.
#
# Copy this template inside bronze_city_prestige_add_building_points:
#
#	if = {
#		limit = {
#			has_building = building_type:your_building_id
#		}
#		change_variable = {
#			name = bronze_city_prestige_score
#			add = 25
#		}
#	}
#
# Buildings are evaluated independently. If several listed buildings coexist
# in one location, their points are cumulative.
# ============================================================================

bronze_city_prestige_add_building_points = {
	# Local exchange and market infrastructure.
	if = {
		limit = {
			has_building = building_type:market_village
		}
		change_variable = {
			name = bronze_city_prestige_score
			add = 10
		}
	}

	if = {
		limit = {
			has_building = building_type:marketplace
		}
		change_variable = {
			name = bronze_city_prestige_score
			add = 20
		}
	}

	if = {
		limit = {
			has_building = building_type:entrepot
		}
		change_variable = {
			name = bronze_city_prestige_score
			add = 30
		}
	}

	if = {
		limit = {
			has_building = building_type:trading_hub
		}
		change_variable = {
			name = bronze_city_prestige_score
			add = 40
		}
	}

	if = {
		limit = {
			has_building = building_type:stock_exchange
		}
		change_variable = {
			name = bronze_city_prestige_score
			add = 50
		}
	}

	# Major Bronze Age administrative and monumental center.
	if = {
		limit = {
			has_building = building_type:bronze_age_palatial_complex
		}
		change_variable = {
			name = bronze_city_prestige_score
			add = 100
		}
	}

	# Historical monuments. Keep every fixed award in this single effect:
	# setup, events, and building on_built blocks must not add it again.
	# Giza is a dedicated Bronze Era building so its effects remain balanced and
	# independent from the aggressive vanilla work-of-art quality multipliers.
	if = {

	if = {
		limit = { has_building = building_type:bronze_monument_abu_simbel }
		change_variable = { name = bronze_city_prestige_score add = 90 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_theban_necropolis }
		change_variable = { name = bronze_city_prestige_score add = 100 }
	}
	if = {

		change_variable = { name = bronze_city_prestige_score add = 130 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_tiryns_cyclopean }
		change_variable = { name = bronze_city_prestige_score add = 80 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_pylos_palace }
		change_variable = { name = bronze_city_prestige_score add = 90 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_troy_citadel }
		change_variable = { name = bronze_city_prestige_score add = 110 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_knossos_complex }
		change_variable = { name = bronze_city_prestige_score add = 120 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_ugarit_complex }
		change_variable = { name = bronze_city_prestige_score add = 100 }

		limit = { has_building = building_type:bronze_monument_ur_ziggurat }

	}
	if = {
		limit = { has_building = building_type:bronze_monument_assur_complex }
		change_variable = { name = bronze_city_prestige_score add = 110 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_chogha_zanbil }
		change_variable = { name = bronze_city_prestige_score add = 100 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_yin_xu }

		change_variable = { name = bronze_city_prestige_score add = 90 }
	}

	if = {
		limit = {
			OR = {
				has_building = building_type:bronze_monument_stonehenge
				any_work_of_art_in_location = {
					this = work_of_art:stonehenge
					is_art_destroyed = no
				}
			}
		}
		change_variable = { name = bronze_city_prestige_score add = 90 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_avebury }
		change_variable = { name = bronze_city_prestige_score add = 70 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_bru_na_boinne }
		change_variable = { name = bronze_city_prestige_score add = 90 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_great_pyramid_new }
		change_variable = { name = bronze_city_prestige_score add = 200 }
	}
	if = {
		limit = { has_building = building_type:bronze_monument_great_pyramid_unfinished }
		change_variable = { name = bronze_city_prestige_score add = 35 }
	}
}
