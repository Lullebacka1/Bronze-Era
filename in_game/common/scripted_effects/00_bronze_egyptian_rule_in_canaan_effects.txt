egyptian_rule_in_canaan_begin = {
	custom_description = {
		text = egyptian_rule_in_canaan_begin_text
		if = {
			limit = {
				situation:egyptian_rule_in_canaan = {
					situation_is_active = no
					situation_has_ended = no
				}
			}
			activate_situation = situation:egyptian_rule_in_canaan
		}

		situation:egyptian_rule_in_canaan ?= {
			set_variable = { name = egyptian_authority_in_canaan value = 80 }
			set_variable = { name = egyptian_rule_in_canaan_phase value = 1 }
			set_variable = { name = egyptian_rule_in_canaan_months value = 0 }
			set_variable = { name = egyptian_rule_in_canaan_annual_counter value = 12 }
			set_variable = { name = egyptian_rule_in_canaan_active value = 1 }
			egyptian_rule_in_canaan_refresh_subject_count = yes
		}

		set_variable = { name = egyptian_rule_in_canaan_overlord value = 1 }

		every_country = {
			limit = { egyptian_rule_in_canaan_city_kingdom = yes }
			set_variable = { name = egyptian_rule_in_canaan_subject_member value = 1 }
		}
	}
}

egyptian_rule_in_canaan_add_authority = {
	custom_description = {
		text = egyptian_rule_in_canaan_add_authority_text
		value = $value$
		situation:egyptian_rule_in_canaan ?= {
			if = {
				limit = { NOT = { has_variable = egyptian_authority_in_canaan } }
				set_variable = { name = egyptian_authority_in_canaan value = 80 }
			}
			change_variable = { name = egyptian_authority_in_canaan add = $value$ }
			if = {
				limit = { var:egyptian_authority_in_canaan > 100 }
				set_variable = { name = egyptian_authority_in_canaan value = 100 }
			}
			if = {
				limit = { var:egyptian_authority_in_canaan < 0 }
				set_variable = { name = egyptian_authority_in_canaan value = 0 }
			}
		}
	}
}

				limit = { var:egyptian_authority_in_canaan >= 30 }

			}
		}
	}
}

egyptian_rule_in_canaan_annual_pressure = {
	custom_description = {
		text = egyptian_rule_in_canaan_annual_pressure_text
		situation:egyptian_rule_in_canaan ?= {
			if = {
				limit = {
					situation:sea_peoples_crisis ?= {
						has_variable = sea_peoples_global_phase
						var:sea_peoples_global_phase >= 3
					}
				}
				egyptian_rule_in_canaan_add_authority = { value = -2 }
			}
			if = {
				limit = {
					situation:sea_peoples_crisis ?= {
						has_variable = sea_peoples_global_phase
						var:sea_peoples_global_phase >= 4
					}
				}
				egyptian_rule_in_canaan_add_authority = { value = -2 }
			}
			if = {
				limit = {
					situation:sea_peoples_crisis ?= {
						has_variable = sea_peoples_global_pressure
						var:sea_peoples_global_pressure >= 70
					}
				}
				egyptian_rule_in_canaan_add_authority = { value = -2 }
			}
		}

		c:0002G ?= {
			if = {
				limit = { is_during_bankruptcy = yes }
				egyptian_rule_in_canaan_add_authority = { value = -5 }
			}
			if = {
				limit = { stability <= 0 }
				egyptian_rule_in_canaan_add_authority = { value = -4 }
			}
			if = {
				limit = { is_at_war = yes }
				egyptian_rule_in_canaan_add_authority = { value = -1 }
			}
		}

		egyptian_rule_in_canaan_refresh_restless_subjects = yes
		situation:egyptian_rule_in_canaan ?= {
			if = {
				limit = { var:egyptian_rule_in_canaan_restless_subjects >= 4 }
				egyptian_rule_in_canaan_add_authority = { value = -4 }
				if = {
					limit = { NOT = { has_variable = egyptian_rule_tribute_refusal_recent } }
					set_variable = { name = egyptian_rule_tribute_refusal_recent value = 1 years = 5 }
					c:0002G ?= { trigger_event_non_silently = egyptian_rule_in_canaan.10 }
				}
			}
		}

		egyptian_rule_in_canaan_check_garrison_loss = yes
	}
}
