# Bronze Era - Dynamic Ruins
# Location scope unless stated otherwise.
#
# Ruin stats are generated once when a city becomes a ruin. The GUI then reads
# persistent variables from the location, avoiding map-wide GUI scans.

be_ruins_generate_stats_effect = {
	custom_description = {
		text = be_ruins_generate_stats_effect_tt

		remove_variable = be_ruins_type_generic
		remove_variable = be_ruins_type_capital
		remove_variable = be_ruins_type_palace
		remove_variable = be_ruins_type_temple
		remove_variable = be_ruins_type_port
		remove_variable = be_ruins_type_fortress
		remove_variable = be_ruins_type_village

		set_variable = { name = be_ruins_stats_generated value = 1 }
		set_variable = { name = be_ruins_importance_score value = 20 }
		set_variable = { name = be_ruins_type_generic value = 1 }

		if = {
			limit = { location_rank = location_rank:megalopolis }
			change_variable = { name = be_ruins_importance_score add = 70 }
		}
		else_if = {
			limit = { location_rank = location_rank:city }
			change_variable = { name = be_ruins_importance_score add = 45 }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			change_variable = { name = be_ruins_importance_score add = 20 }
		}

		if = {
			limit = {
				has_variable = bronze_ruins_former_prestige
				var:bronze_ruins_former_prestige >= 250
			}
			change_variable = { name = be_ruins_importance_score add = 90 }
		}
		else_if = {
			limit = {
				has_variable = bronze_ruins_former_prestige
				var:bronze_ruins_former_prestige >= 150
			}
			change_variable = { name = be_ruins_importance_score add = 70 }
		}
		else_if = {
			limit = {
				has_variable = bronze_ruins_former_prestige
				var:bronze_ruins_former_prestige >= 75
			}
			change_variable = { name = be_ruins_importance_score add = 45 }
		}
		else_if = {
			limit = {
				has_variable = bronze_ruins_former_prestige
				var:bronze_ruins_former_prestige >= 25
			}
			change_variable = { name = be_ruins_importance_score add = 20 }
		}

		if = {
			limit = { is_capital = yes }
			remove_variable = be_ruins_type_generic
			set_variable = { name = be_ruins_type_capital value = 1 }
			change_variable = { name = be_ruins_importance_score add = 45 }
		}

		if = {
			limit = {
				OR = {
					has_building = building_type:bronze_age_palatial_complex
					has_building = building_type:bronze_age_palace_house
					has_building = building_type:palace_registry
					has_building = building_type:palace_trade_registry
				}
			}
			remove_variable = be_ruins_type_generic
			set_variable = { name = be_ruins_type_palace value = 1 }
			change_variable = { name = be_ruins_importance_score add = 55 }
		}

		if = {
			limit = {
				OR = {
					has_building = building_type:temple_complex
					has_building = building_type:temple_granary
					has_building = building_type:temple_scribes_guild
					has_building = building_type:temple_healing_sanctuary
					has_building = building_type:temple_court
					has_building = building_type:megalithic_religious_monument
				}
			}
			remove_variable = be_ruins_type_generic
			set_variable = { name = be_ruins_type_temple value = 1 }
			change_variable = { name = be_ruins_importance_score add = 35 }
		}

		if = {
			limit = {
				OR = {
					is_port = yes
					has_building = building_type:trading_hub
					has_building = building_type:entrepot
				}
			}
			remove_variable = be_ruins_type_generic
			set_variable = { name = be_ruins_type_port value = 1 }
			change_variable = { name = be_ruins_importance_score add = 30 }
		}

		if = {
			limit = {
				OR = {
					has_building = building_type:stockade
					has_building = building_type:bronze_city_walls
					has_building = building_type:hill_fort
				}
			}
			remove_variable = be_ruins_type_generic
			set_variable = { name = be_ruins_type_fortress value = 1 }
			change_variable = { name = be_ruins_importance_score add = 25 }
		}

		if = {
			limit = {
				NOT = { has_variable = be_ruins_type_capital }
				NOT = { has_variable = be_ruins_type_palace }
				NOT = { has_variable = be_ruins_type_temple }
				NOT = { has_variable = be_ruins_type_port }
				NOT = { has_variable = be_ruins_type_fortress }
				OR = {
					location_rank = location_rank:rural_settlement
					location_rank = location_rank:town
				}
			}
			remove_variable = be_ruins_type_generic
			set_variable = { name = be_ruins_type_village value = 1 }
		}

		random_list = {
			20 = { change_variable = { name = be_ruins_importance_score add = -10 } }
			25 = { change_variable = { name = be_ruins_importance_score add = 0 } }
			25 = { change_variable = { name = be_ruins_importance_score add = 10 } }
			20 = { change_variable = { name = be_ruins_importance_score add = 20 } }
			10 = { change_variable = { name = be_ruins_importance_score add = 35 } }
		}

		if = {
			limit = { var:be_ruins_importance_score >= 220 }
			set_variable = { name = be_ruins_difficulty value = 5 }
			set_variable = { name = be_ruins_risk value = 5 }
			set_variable = { name = be_ruins_reward_value value = 5 }
			set_variable = { name = be_ruins_artifact_chance value = 45 }
			set_variable = { name = be_ruins_gold_potential value = 180 }
			set_variable = { name = be_ruins_knowledge_potential value = 20 }
			set_variable = { name = be_ruins_lore_value value = 5 }
			set_variable = { name = be_ruins_danger_level value = 5 }
			set_variable = { name = be_ruins_required_manpower value = 300 }
			set_variable = { name = be_ruins_required_gold value = 120 }
			set_variable = { name = be_ruins_required_goods value = 4 }
			set_variable = { name = be_ruins_duration_months value = 12 }
			set_variable = { name = be_ruins_success_chance value = 40 }
		}
		else_if = {
			limit = { var:be_ruins_importance_score >= 160 }
			set_variable = { name = be_ruins_difficulty value = 4 }
			set_variable = { name = be_ruins_risk value = 4 }
			set_variable = { name = be_ruins_reward_value value = 4 }
			set_variable = { name = be_ruins_artifact_chance value = 30 }
			set_variable = { name = be_ruins_gold_potential value = 120 }
			set_variable = { name = be_ruins_knowledge_potential value = 15 }
			set_variable = { name = be_ruins_lore_value value = 4 }
			set_variable = { name = be_ruins_danger_level value = 4 }
			set_variable = { name = be_ruins_required_manpower value = 220 }
			set_variable = { name = be_ruins_required_gold value = 100 }
			set_variable = { name = be_ruins_required_goods value = 3 }
			set_variable = { name = be_ruins_duration_months value = 9 }
			set_variable = { name = be_ruins_success_chance value = 50 }
		}
		else_if = {
			limit = { var:be_ruins_importance_score >= 100 }
			set_variable = { name = be_ruins_difficulty value = 3 }
			set_variable = { name = be_ruins_risk value = 3 }
			set_variable = { name = be_ruins_reward_value value = 3 }
			set_variable = { name = be_ruins_artifact_chance value = 20 }
			set_variable = { name = be_ruins_gold_potential value = 80 }
			set_variable = { name = be_ruins_knowledge_potential value = 10 }
			set_variable = { name = be_ruins_lore_value value = 3 }
			set_variable = { name = be_ruins_danger_level value = 3 }
			set_variable = { name = be_ruins_required_manpower value = 150 }
			set_variable = { name = be_ruins_required_gold value = 75 }
			set_variable = { name = be_ruins_required_goods value = 2 }
			set_variable = { name = be_ruins_duration_months value = 6 }
			set_variable = { name = be_ruins_success_chance value = 60 }
		}
		else_if = {
			limit = { var:be_ruins_importance_score >= 55 }
			set_variable = { name = be_ruins_difficulty value = 2 }
			set_variable = { name = be_ruins_risk value = 2 }
			set_variable = { name = be_ruins_reward_value value = 2 }
			set_variable = { name = be_ruins_artifact_chance value = 10 }
			set_variable = { name = be_ruins_gold_potential value = 50 }
			set_variable = { name = be_ruins_knowledge_potential value = 5 }
			set_variable = { name = be_ruins_lore_value value = 2 }
			set_variable = { name = be_ruins_danger_level value = 2 }
			set_variable = { name = be_ruins_required_manpower value = 100 }
			set_variable = { name = be_ruins_required_gold value = 50 }
			set_variable = { name = be_ruins_required_goods value = 1 }
			set_variable = { name = be_ruins_duration_months value = 4 }
			set_variable = { name = be_ruins_success_chance value = 70 }
		}
		else = {
			set_variable = { name = be_ruins_difficulty value = 1 }
			set_variable = { name = be_ruins_risk value = 1 }
			set_variable = { name = be_ruins_reward_value value = 1 }
			set_variable = { name = be_ruins_artifact_chance value = 5 }
			set_variable = { name = be_ruins_gold_potential value = 30 }
			set_variable = { name = be_ruins_knowledge_potential value = 2 }
			set_variable = { name = be_ruins_lore_value value = 1 }
			set_variable = { name = be_ruins_danger_level value = 1 }
			set_variable = { name = be_ruins_required_manpower value = 75 }
			set_variable = { name = be_ruins_required_gold value = 35 }
			set_variable = { name = be_ruins_required_goods value = 1 }
			set_variable = { name = be_ruins_duration_months value = 3 }
			set_variable = { name = be_ruins_success_chance value = 80 }
		}

		if = {
			limit = { has_variable = be_ruins_type_temple }
			change_variable = { name = be_ruins_artifact_chance add = 10 }
			change_variable = { name = be_ruins_lore_value add = 1 }
		}
		if = {
			limit = { has_variable = be_ruins_type_palace }
			change_variable = { name = be_ruins_gold_potential add = 40 }
			change_variable = { name = be_ruins_knowledge_potential add = 5 }
		}
		if = {
			limit = { has_variable = be_ruins_type_port }
			change_variable = { name = be_ruins_gold_potential add = 25 }
		}
		if = {
			limit = { has_variable = be_ruins_type_fortress }
			change_variable = { name = be_ruins_danger_level add = 1 }
			change_variable = { name = be_ruins_risk add = 1 }
		}
	}
}

# Country scope. Requires scope:be_ruin_location.
be_ruins_pay_selected_expedition_cost_effect = {
	if = {
		limit = { scope:be_ruin_location ?= { var:be_ruins_difficulty >= 5 } }
		add_gold = -120
		add_manpower_pool = -300
	}
	else_if = {
		limit = { scope:be_ruin_location ?= { var:be_ruins_difficulty >= 4 } }
		add_gold = -100
		add_manpower_pool = -220
	}
	else_if = {
		limit = { scope:be_ruin_location ?= { var:be_ruins_difficulty >= 3 } }
		add_gold = -75
		add_manpower_pool = -150
	}
	else_if = {
		limit = { scope:be_ruin_location ?= { var:be_ruins_difficulty >= 2 } }
		add_gold = -50
		add_manpower_pool = -100
	}
	else = {
		add_gold = -35
		add_manpower_pool = -75
	}
}

# Country scope. Requires scope:bronze_ruins_site when possible.
be_ruins_dynamic_gold_reward_effect = {
	if = {
		limit = { scope:bronze_ruins_site ?= { var:be_ruins_reward_value >= 5 } }
		add_gold = 220
		add_prestige = prestige_severe_bonus
	}
	else_if = {
		limit = { scope:bronze_ruins_site ?= { var:be_ruins_reward_value >= 4 } }
		add_gold = 150
		add_prestige = prestige_mild_bonus
	}
	else_if = {
		limit = { scope:bronze_ruins_site ?= { var:be_ruins_reward_value >= 3 } }
		add_gold = 100
		add_prestige = prestige_weak_bonus
	}
	else_if = {
		limit = { scope:bronze_ruins_site ?= { var:be_ruins_reward_value >= 2 } }
		add_gold = 60
		add_prestige = prestige_weak_bonus
	}
	else = {
		add_gold = 35
	}
}

be_ruins_dynamic_knowledge_reward_effect = {
	if = {
		limit = { scope:bronze_ruins_site ?= { var:be_ruins_knowledge_potential >= 15 } }
		add_research_progress = research_progress_extreme_bonus
		add_prestige = prestige_mild_bonus
	}
	else_if = {
		limit = { scope:bronze_ruins_site ?= { var:be_ruins_knowledge_potential >= 10 } }
		add_research_progress = research_progress_severe_bonus
		add_prestige = prestige_weak_bonus
	}
	else_if = {
		limit = { scope:bronze_ruins_site ?= { var:be_ruins_knowledge_potential >= 5 } }
		add_research_progress = research_progress_mild_bonus
	}
	else = {
		add_research_progress = research_progress_weak_bonus
	}
}

be_ruins_dynamic_negative_reward_effect = {
	if = {
		limit = { scope:bronze_ruins_site ?= { var:be_ruins_risk >= 4 } }
		add_gold = -60
		add_prestige = prestige_mild_penalty
		add_stability = stability_weak_penalty
	}
	else_if = {
		limit = { scope:bronze_ruins_site ?= { var:be_ruins_risk >= 3 } }
		add_gold = -40
		add_prestige = prestige_weak_penalty
	}
	else = {
		add_gold = -20
	}
}
