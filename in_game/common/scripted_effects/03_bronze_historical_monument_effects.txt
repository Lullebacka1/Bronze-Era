# Location scope. Native monument works of art are not buildings, so the
# generic building-destruction loop cannot remove them. Run this only after
# the monument snapshot and only when the location becomes permanent ruins.
bronze_historic_monuments_destroy_native_art_for_ruins = {
	if = {
		limit = {
			any_work_of_art_in_location = {
				this = work_of_art:pyramid_art_key
				is_art_destroyed = no
			}
		}
		random_work_of_art_in_location = {
			limit = { this = work_of_art:pyramid_art_key }
			destroy_art = this
		}
	}
	if = {
		limit = {
			any_work_of_art_in_location = {
				this = work_of_art:stonehenge
				is_art_destroyed = no
			}
		}
		random_work_of_art_in_location = {
			limit = { this = work_of_art:stonehenge }
			destroy_art = this
		}
	}
}

# Prepared location effect for Dholavira once its exact EU5 location is known.
bronze_historical_ruin_dholavira_setup_effect = {
	set_variable = { name = bronze_ancient_ruins value = 1 }
	set_variable = { name = bronze_ruins_destroyed_year value = 1 }
	set_variable = { name = bronze_ruins_former_prestige value = 160 }
	set_variable = { name = bronze_ruins_historical_culture value = culture:harappan }
	set_variable = { name = bronze_ruins_historical_religion value = religion:harappan }
	be_ruins_generate_stats_effect = yes
	remove_variable = be_ruins_type_generic
	remove_variable = be_ruins_type_village
	set_variable = { name = be_ruins_type_palace value = 1 }
	set_variable = { name = be_ruins_type_port value = 1 }
	set_variable = { name = bronze_ruins_had_royal_monument value = 1 }
	set_variable = { name = bronze_ruins_had_commercial_monument value = 1 }
	add_location_modifier = {
		modifier = bronze_ancient_ruins_location
		years = -1
		mode = replace
	}
}
