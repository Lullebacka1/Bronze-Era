# Bronze Era Canaanite and Peleset regional kingdoms.
# These are small kingdom-tier formables: they require independence and direct
# ownership of every listed location. No generic "Canaanite" tag is created.

ISRAEL_f = {
	level = 3
	required_locations_fraction = 1.00
	capital_required = no
	potential_requires_own = yes
	rule = plausible
	content_priority = 760

	potential = {
		OR = {
			culture = culture:canaanite
			tag = HACOR
			tag = SICHE
			tag = GEZE
			tag = PEHEL
		}
		NOT = { tag = ISRAEL }
		NOT = { country_exists = c:ISRAEL }
		NOT = { has_variable = bronze_formed_canaanite_kingdom }
	}

	allow = {
		is_subject = no
		religion = religion:canaanite
		owns = location:safed
		owns = location:nablus
		owns = location:jaffa
		owns = location:irbid
	}

	name = ISRAEL
	flag = ISRAEL
	adjective = ISRAEL_ADJ
	tag = ISRAEL
	color = map_ISRAEL

	locations = {
		safed nablus jaffa irbid
	}

	form_effect = {
		change_culture = culture:canaanite
		change_religion = religion:canaanite
		set_capital = location:nablus
		add_accepted_culture = culture:peleset
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}

		set_variable = { name = bronze_formed_canaanite_kingdom value = 1 }
		add_legitimacy = legitimacy_weak_bonus
		add_prestige = prestige_weak_bonus
		add_stability = stability_weak_bonus
		add_country_modifier = {
			modifier = bronze_canaanite_kingdom_foundation
			years = -1
			mode = add_and_extend
		}
	}
}
