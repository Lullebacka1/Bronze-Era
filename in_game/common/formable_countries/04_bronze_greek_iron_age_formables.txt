# Bronze Era Greek Iron Age regional formables.
# These require Iron Age or later, independence, and full control of their listed
# locations. Mycenae itself is excluded so the old palace hegemon does not use
# later northern Greek regional formations.

	level = 3
	required_locations_fraction = 1.00
	capital_required = no
	potential_requires_own = yes
	rule = plausible
	content_priority = 715

	potential = {
		OR = {
			culture = { has_culture_group = culture_group:aegean_bronze_group }
			culture = { has_culture_group = culture_group:greek_group }
			religion = religion:hellenic
		}
		NOT = { tag = 0001G }
		NOT = { tag = MYC }
		NOT = { tag = MACED }
		NOT = { country_exists = c:MACED }
		NOT = { has_variable = bronze_formed_greek_iron_realm }
	}

	allow = {

		owns = location:ohrid
		owns = location:ostrovo
		owns = location:polygyros
		owns = location:prilep
		owns = location:prosek
		owns = location:rentina
		owns = location:serres
		owns = location:servia
		owns = location:siatista
		owns = location:thessaloniki
		owns = location:veria
	}

	name = MACED
	flag = MACED
	adjective = MACED_ADJ
	tag = MACED
	color = map_MACED

	locations = {
		bitola drama edessa florina giannitsa grevena gynaikokastro hatera
		kassandreia kastoria kavala kilkis mount_athos ohrid ostrovo
		polygyros prilep prosek rentina serres servia siatista
		thessaloniki veria
	}

	form_effect = {
		bronze_formable_make_monarchy_if_tribal = yes
		change_government_type = government_type:monarchy
		change_culture = culture:macedonian
		change_religion = religion:hellenic
		set_capital = location:edessa
		add_accepted_culture = culture:mycenaean
		add_accepted_culture = culture:paeonian
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}

		set_variable = { name = bronze_formed_greek_iron_realm value = 1 }
		add_legitimacy = legitimacy_weak_bonus
		add_prestige = prestige_weak_bonus
		add_stability = stability_weak_bonus
		add_country_modifier = {
			modifier = bronze_macedonian_iron_kingship
			years = -1
			mode = add_and_extend
		}
	}
}
