mycenaean_civil_war = {
	monthly_spawn_chance = 1

	can_start = {
		has_global_variable = { 
			name = trojan_war_phase_2_ended 
			value = yes 
		}
		has_global_variable = { 
			NOT = { mycenaean_civil_war_ended } }
		international_organization:mycenaean_league = {
			has_variable = mycenaean_league_authority
			var:mycenaean_league_authority <= 5 
		}
	}

	modifier = {
		
	}

	can_end = {
		OR = {
			current_year > 1500
			var:target_var >= 90
		}
	}

	on_start = {
		# start the civil war 
	}

	on_monthly = {
		# disater event for cival war
		random_list = {
			11 = { }
			1 = {
				fire_example_disaster_event = yes
			}
		}
	}

	on_end = {
		
		set_global_variable = { name = mycenaean_civil_war_ended value = yes }
	}
}
