patrimonialism_reform = {#https://en.wikipedia.org/wiki/Patrimonialism
	age = age_1_traditions
	
	allow = {#It is not meant for highly organized countries
		country_rank_level <= 2
		is_country_religion_pagan = yes
	}

	locked = {
		capital = {
			OR = {
				num_pop_type:nobles < {
					value = population
					divide = 100
					multiply = 1
				}
				num_pop_type:clergy < {
					value = population
					divide = 100
					multiply = 2
				} 
				num_pop_type:burghers < {
					value = population
					divide = 100
					multiply = 5
				}
			}
		}
	}

	country_modifier = {

		#Values
		monthly_towards_traditionalist = societal_value_large_monthly_move

		#Positives
		global_estate_power = -0.5

		#Negatives
		tax_income_efficiency = large_tax_income_efficiency_penalty
		government_size = -1
		global_estate_target_satisfaction = large_permanent_target_satisfaction_penalty
	}

	on_activate = {
		set_parliament_type = parliament_type:assembly
	}

	years = 2
}
