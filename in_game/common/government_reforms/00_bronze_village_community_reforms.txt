hereditary_chiefdom_reform = {
	age = age_1_traditions
	major = yes
	unique = yes
	government = village_community

	potential = {
		government_type = government_type:village_community
	}
	allow = {
		num_locations >= 3
		prestige >= 10
		stability >= 0
		societal_value:centralization_vs_decentralization < 80
	}

	on_activate = {
		change_government_type = government_type:tribe
		change_heir_selection = heir_selection:tribal_oldest_male
		set_parliament_type = parliament_type:assembly
		set_societal_value = { value = 50 type = centralization_vs_decentralization }
		if = {
			limit = { country_has_estate = estate_type:nobles_estate }
			add_estate_satisfaction = { type = estate_type:nobles_estate value = large_permanent_target_satisfaction }
		}
	}

	years = 5
}

village_community_adopt_monarchy_reform = {
	age = age_1_traditions
	major = yes
	unique = yes
	government = village_community

	potential = {
		government_type = government_type:village_community
	}
	allow = {
		num_locations >= 10
		country_rank_level >= 2
		prestige >= 25
		stability >= 0
		societal_value:centralization_vs_decentralization <= -25
	}

	on_activate = {
		change_government_type = government_type:monarchy
		change_heir_selection = heir_selection:cognatic_primogeniture
		set_parliament_type = parliament_type:no_parliament
		set_societal_value = { value = -50 type = centralization_vs_decentralization }
		add_government_power = government_power_extreme_bonus
		if = {
			limit = { country_has_estate = estate_type:nobles_estate }
			add_estate_satisfaction = { type = estate_type:nobles_estate value = large_permanent_target_satisfaction }
		}
	}

	years = 10
}
