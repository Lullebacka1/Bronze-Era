# Root = country. Each owned location is considered at most once per year.
bronze_city_prestige_runtime_bootstrap = {
	trigger = {
		is_real_country = yes
		NOT = {
			has_global_variable = bronze_city_prestige_runtime_v9_initialized
		}
	}

	effect = {
		# Runs once on the first monthly pulse, or later for converted saves
		# where the runtime marker is missing.
		remove_global_variable = bronze_city_prestige_manager
		bronze_city_prestige_initialize_world = yes
	}
}

# Root = country. Runs once per campaign version for the human player.
# Version 17 refreshes the ranking cache after expanding the City Prestige
# tooltip with score breakdowns and neighboring ranked cities.
bronze_city_prestige_player_rank_bootstrap = {
	trigger = {
		is_real_country = yes
		is_ai = no
		NOT = {
			has_variable = bronze_city_prestige_rank_v17_initialized
		}
	}

	effect = {
		# Repair saves where the runtime marker survived but the world registry
		# did not.
		if = {
			limit = {
				OR = {
					NOT = {
						has_global_variable_list = bronze_city_prestige_world_cities
					}
					AND = {
						has_global_variable_list = bronze_city_prestige_world_cities
						global_variable_list_size = {
							name = bronze_city_prestige_world_cities
							value < 1
						}
					}
				}
			}
			remove_global_variable = bronze_city_prestige_manager
			bronze_city_prestige_initialize_world = yes
		}

		# Restore the former behavior: the human bootstrap performs one complete
		# world ranking pass so valid cities cannot remain without World Rank.
		bronze_city_prestige_update_world_ranking = yes

		set_variable = {
			name = bronze_city_prestige_rank_v17_initialized
			value = 1
		}
	}
}

# Root = human country. Cheap monthly guard for converted or damaged saves.
bronze_city_prestige_player_rank_repair = {
	trigger = {
		is_real_country = yes
		is_ai = no
		OR = {
			NOT = {
				has_global_variable = bronze_city_prestige_manager
			}
			NOT = {
				has_global_variable_list = bronze_city_prestige_world_cities
			}
			AND = {
				has_global_variable_list = bronze_city_prestige_world_cities
				global_variable_list_size = {
					name = bronze_city_prestige_world_cities
					value < 1
				}
			}
			any_owned_location = {
				bronze_city_prestige_is_urban = yes
				NOT = { has_variable = bronze_city_prestige_global_rank }
			}
			any_owned_location = {
				bronze_city_prestige_is_urban = yes
				has_variable = bronze_city_prestige_global_rank
				var:bronze_city_prestige_global_rank < 1
			}
		}
	}

	effect = {
		if = {
			limit = {
				OR = {
					NOT = {
						has_global_variable_list = bronze_city_prestige_world_cities
					}
					AND = {
						has_global_variable_list = bronze_city_prestige_world_cities
						global_variable_list_size = {
							name = bronze_city_prestige_world_cities
							value < 1
						}
					}
				}
			}
			remove_global_variable = bronze_city_prestige_manager
			bronze_city_prestige_initialize_world = yes
		}
		else = {
			if = {
				limit = {
					NOT = { has_global_variable = bronze_city_prestige_manager }
				}
				set_global_variable = {
					name = bronze_city_prestige_manager
					value = this
				}
			}

			bronze_city_prestige_update_world_ranking = yes
		}
	}
}

# Root = country. One-time compatibility bootstrap for campaigns saved before
# the City Prestige system was installed.
bronze_city_prestige_yearly_pulse = {
	trigger = {
		is_real_country = yes
	}

	effect = {
		if = {
			limit = {
				NOT = { has_variable = bronze_city_prestige_update_years }
			}
			set_variable = { name = bronze_city_prestige_update_years value = 0 }
		}
		change_variable = { name = bronze_city_prestige_update_years add = 1 }
		set_variable = { name = bronze_city_prestige_next_update_years value = 5 }
		change_variable = {
			name = bronze_city_prestige_next_update_years
			subtract = var:bronze_city_prestige_update_years
		}

		if = {
			limit = {
				NOT = { has_variable = bronze_city_prestige_eligibility_years }
			}
			set_variable = { name = bronze_city_prestige_eligibility_years value = 0 }
		}

		# Recover cleanly if a converted save predates the manager variable or
		# another script deliberately clears it.
		if = {
			limit = {
				NOT = { has_global_variable = bronze_city_prestige_manager }
			}
			set_global_variable = {
				name = bronze_city_prestige_manager
				value = this
			}
		}

		# Full prestige refresh runs every five yearly pulses. New campaigns
		# still receive their first full ranking during game-start/bootstrap.
		if = {
			limit = { var:bronze_city_prestige_update_years >= 5 }
			set_variable = { name = bronze_city_prestige_update_years value = 0 }
			set_variable = { name = bronze_city_prestige_next_update_years value = 5 }
			set_variable = { name = bronze_city_prestige_eligibility_years value = 0 }
			bronze_city_prestige_yearly_update = yes
			bronze_city_prestige_five_year_eligibility_update = yes

			# Only the designated manager performs the world sort and player-list
			# rebuild, avoiding one global scan per country.
			if = {
				limit = {
					has_global_variable = bronze_city_prestige_manager
					this = global_var:bronze_city_prestige_manager
				}
				set_global_variable = {
					name = bronze_city_prestige_run_five_year_content
					value = 1
				}
				bronze_city_prestige_update_world_ranking = yes
				remove_global_variable = bronze_city_prestige_run_five_year_content
			}
		}
	}
}

# Root = country. One urban story is sampled at most once per yearly pulse,
# with a large no-event weight and a three-year country cooldown.
bronze_city_prestige_urban_event_pulse = {
	trigger = {
		is_real_country = yes
		NOT = { has_variable = bronze_city_prestige_event_cooldown }
		any_owned_location = {
			bronze_city_prestige_is_regional_or_higher = yes
		}
	}

	random_events = {
		sample_count = 1

		250 = 0
		5 = bronze_city_prestige.1
		5 = bronze_city_prestige.2
		5 = bronze_city_prestige.3
		5 = bronze_city_prestige.4
		5 = bronze_city_prestige.5
		5 = bronze_city_prestige.6
		5 = bronze_city_prestige.7
	}
}
