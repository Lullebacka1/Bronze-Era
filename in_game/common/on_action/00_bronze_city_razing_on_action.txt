# Bronze Era - City Sacking / Razing / Ruins
# Called from _hardcoded.txt on_siege_won and on_location_occupied.

bronze_city_razing_on_city_capture = {
	trigger = {
		is_real_country = yes
		scope:target = {
			bronze_city_razing_valid_capture_target = yes
		}
	}

	effect = {
		bronze_city_capture_prepare_scopes = yes

		if = {
			limit = { is_ai = yes }
			if = {
				limit = {
					scope:target = {
						bronze_city_ai_should_preserve_capture_target = yes
					}
				}
				bronze_city_capture_occupy_effect = yes
			}
			else = {
				random_list = {
					50 = { bronze_city_capture_occupy_effect = yes }
					35 = { bronze_city_start_sack_effect = yes }
					10 = { bronze_city_start_raze_effect = yes }
					10 = {
						trigger = { gold < 20 }
						bronze_city_start_sack_effect = yes
					}
					10 = {
						trigger = {
							scope:target = {
								bronze_city_ai_high_value_plunder_target = yes
							}
						}
						bronze_city_start_sack_effect = yes
					}
					5 = {
						trigger = {
							scope:target = {
								bronze_city_ai_high_value_plunder_target = yes
								OR = {
									has_location_modifier = bronze_city_recently_sacked
									has_location_modifier = bronze_city_razed
									has_location_modifier = recently_raided_by_horde
								}
							}
						}
						bronze_city_start_raze_effect = yes
					}
				}
			}
		}
		else = {
			trigger_event_non_silently = {
				id = bronze_city_razing.1
			}
		}
	}
}

# Country scope. Rare AI/player prompt to explore long-abandoned ruins.
bronze_city_ruins_yearly_pulse = {
	trigger = {
		is_real_country = yes
	}

	effect = {
		# Repair saves created before completed razes always became persistent
		# ruins. This also prevents a visible "razed city" state without the
		# ruins button in the location window.
		every_owned_location = {
			limit = {
				has_location_modifier = bronze_city_razed
				NOT = { has_variable = bronze_ancient_ruins }
			}
			bronze_city_create_ancient_ruins_effect = yes
		}

		if = {
			limit = {
				NOT = { has_variable = bronze_ruins_expedition_country_cooldown }
				bronze_city_razing_can_fund_expedition = yes
			}
			random_list = {
				95 = { }
				5 = {
					trigger_event_non_silently = {
						id = bronze_city_razing.9
					}
				}
			}
		}
	}
}
