on_game_start = {
    on_actions = {
        trojan_war_raid_counters_init
    }
}

trojan_war_raid_counters_init = {
    effect = {
        set_global_variable = {
            name = anatolia_raid_success_count
            value = 0
        }
        set_global_variable = {
            name = anatolia_raid_failed_count
            value = 0
        }
    }
}
