monthly_country_pulse = {
	on_actions = {
		mycenaean_influence_pulse
		monthly_update_mycenaean_one
		monthly_update_mycenaean_two
		monthly_update_mycenaean_fragmentation
		mycenaean_league_yearly_pulse
	}
}
mycenaean_league_yearly_pulse = {
	trigger = {
		is_member_of_international_organization = international_organization:mycenaean_league
		international_organization:mycenaean_league = {
			var:mycenaean_league_months >= 72
		}
	}

	random_events = {
		chance_to_happen = 22
		100 = 0
		8 = mycenaean_league.1
		8 = mycenaean_league.2
		7 = mycenaean_league.10
		6 = mycenaean_league.20
		5 = mycenaean_league.30
		4 = mycenaean_league.40
		7 = mycenaean_league.50
		5 = mycenaean_league.60
		4 = mycenaean_league.70
		3 = mycenaean_league.80
		4 = mycenaean_league.90
		4 = mycenaean_league.100
		3 = mycenaean_league.110
		5 = mycenaean_league.120
	}
}

mycenaean_influence_pulse = {
	trigger = {
		OR = {
			current_month = 2
			current_month = 8
		}
	}
	effect = {
		mycenaean_influence = yes
	}
}

monthly_update_mycenaean_one = {
	trigger = {
		international_organization:mycenaean_league = {
			leader_country = root
			has_variable = mycenaean_league_authority
			leader_country = {
				NOT = { has_country_modifier = influence_impact_league_authority_coldown }
			}
		}
	}
	effect = {
		influence_impact_league_authority = yes
	}
}

monthly_update_mycenaean_two = {
	trigger = {
		international_organization:mycenaean_league = {
			leader_country = root
			has_variable = mycenaean_league_authority
			leader_country = {
				NOT = { has_country_modifier = after_trojan_war_impact_league_authority_coldown }
			}
		}
	}
	effect = {
		after_trojan_war_impact_league_authority = yes
	}
}

monthly_update_mycenaean_fragmentation = {
	trigger = {
		international_organization:mycenaean_league = {
			leader_country = root
		}
	}
	effect = {
		mycenaean_fragmentation_monthly_update = yes
	}
}

on_game_start = {
	effect = {
		international_organization:mycenaean_league = {
			if = {
				limit = { NOT = { has_variable = mycenaean_fragmentation_months } }
				set_variable = { name = mycenaean_fragmentation_months value = 0 }
			}
			mycenaean_fragmentation_reevaluate_strongest_palaces = yes
		}
		mycenaean_influence = yes
	}
}
