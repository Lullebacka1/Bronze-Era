namespace = ise_long_dist_trade

ise_long_dist_trade.1 = {
	type = country_event
	title = institution_spawn_events_long_dist_trade.1.title
	desc = institution_spawn_events_long_dist_trade.1.desc
	category = situation_event
	image = "gfx/interface/illustrations/situation/institution_spawn_events_long_dist_trade_1.dds"
	outcome = neutral

	trigger = {
		NOT = {	has_embraced_institution = institution:bronze_routes }
		has_ruler = yes
		country_has_estate = estate_type:burghers_estate

		NOT = {
			var:long_distance_trade_spawning_variable = 1
		}

		any_owned_location = {
			is_market_center = yes
			market = {
				total_goods_value_traded > 150
			}
		}
	}

	immediate = {
		set_variable = {
			name = long_distance_trade_spawning_variable
			value = 1
			years = 3
		}
	}

	option = {
		name = institution_spawn_events_long_dist_trade.1.a

		add_stability = stability_weak_bonus
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_penalty }

		add_country_modifier = {
			modifier = merchants_search_for_markets_modifier
			years = 1
		}
	}

	option = {
		name = institution_spawn_events_long_dist_trade.1.b

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
		change_gold_effect = { scale = -2 }


		add_country_modifier = {
			modifier = merchants_search_for_markets_modifier
			years = 1
		}

		custom_tooltip = {
			text = long_distance_trade_spawns_1_tt
			every_owned_location = {
				limit = {
					market = {
						total_goods_value_traded > 50
					}
				}

				change_institution_progress = { type = institution:bronze_routes value = institution_progress_weak_bonus }
			}
		}

		trigger_event_non_silently = {
			id = ise_long_dist_trade.2
			months = 6
		}
	}

	option = {
		name = institution_spawn_events_long_dist_trade.1.c

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		change_gold_effect = { scale = -5 }

		add_country_modifier = {
			modifier = merchants_search_for_markets_modifier
			years = 1
		}

		custom_tooltip = {
			text = long_distance_trade_spawns_1_tt
			every_owned_location = {
				limit = {
					market = {
						total_goods_value_traded > 50
					}
				}

				change_institution_progress = { type = institution:bronze_routes value = institution_progress_weak_bonus }
			}
		}

		trigger_event_non_silently = {
			id = ise_long_dist_trade.2
			months = 6
		}
	}
	
}
