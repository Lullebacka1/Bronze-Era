egyptian_rule_in_canaan.10 = {
	type = country_event
	title = egyptian_rule_in_canaan.10.title
	desc = egyptian_rule_in_canaan.10.desc
	category = situation_event
	image = "gfx/interface/illustrations/situation/sea_peoples_crisis.dds"
	outcome = negative
	orphan = yes

	trigger = {
		egyptian_rule_in_canaan_egypt = yes
		situation:egyptian_rule_in_canaan ?= {
			has_variable = egyptian_rule_in_canaan_restless_subjects
			var:egyptian_rule_in_canaan_restless_subjects >= 4
		}
	}

	option = {
		name = egyptian_rule_in_canaan.10.a
		custom_tooltip = egyptian_rule_in_canaan_10a_tt
		egyptian_rule_in_canaan_add_authority = { value = -4 }
	}
	option = {
		name = egyptian_rule_in_canaan.10.b
		trigger = {
			gold >= 45
		}
		custom_tooltip = egyptian_rule_in_canaan_10b_tt
		add_gold = -45
		egyptian_rule_in_canaan_add_authority = { value = 3 }
		every_subject = {
			limit = { egyptian_rule_in_canaan_city_kingdom = yes }
			add_liberty_desire = liberty_desire_weak_minus
		}
		ai_chance = {
			factor = 40
			modifier = {
				factor = 0.3
				gold < 80
			}
			modifier = {
				factor = 2
				situation:egyptian_rule_in_canaan ?= { var:egyptian_authority_in_canaan < 45 }
			}
		}
	}
}

egyptian_rule_in_canaan.100 = {
	type = country_event
	title = egyptian_rule_in_canaan.100.title
	desc = egyptian_rule_in_canaan.100.desc
	category = situation_event
	image = "gfx/interface/illustrations/situation/sea_peoples_crisis.dds"
	outcome = negative
	major = yes
	orphan = yes

	trigger = {
		egyptian_rule_in_canaan_egypt = yes
		situation:egyptian_rule_in_canaan ?= {
			var:egyptian_authority_in_canaan <= 0
		}
	}

	option = {
		name = egyptian_rule_in_canaan.100.a
		custom_tooltip = egyptian_rule_in_canaan_100a_tt
		egyptian_rule_in_canaan_withdraw = yes
		ai_chance = {
			factor = 70
			modifier = {
				factor = 2
				is_during_bankruptcy = yes
			}
			modifier = {
				factor = 2
				stability <= 0
			}
			modifier = {
				factor = 0.4
				gold >= 300
				manpower = { value >= 800 }
			}
		}
	}
	option = {
		name = egyptian_rule_in_canaan.100.b
		trigger = {
			NOT = { has_global_variable = egyptian_rule_canaan_must_be_held_used }
			is_during_bankruptcy = no
			gold >= 180
			manpower = { value >= 500 }
		}
		custom_tooltip = egyptian_rule_in_canaan_100b_tt
		egyptian_rule_in_canaan_last_stand = yes
		ai_chance = {
			factor = 30
			modifier = {
				factor = 2
				gold >= 400
				manpower = { value >= 1200 }
				stability >= 30
			}
			modifier = {
				factor = 0.2
				is_at_war = yes
			}
		}
	}
}
