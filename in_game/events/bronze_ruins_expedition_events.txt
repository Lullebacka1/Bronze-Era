# Bronze Era - Narrative Ruins Expeditions
# These events extend the existing expedition backend. Each event expects
# scope:bronze_ruins_site to point at the ruin location.

bronze_city_razing.20 = {
	type = country_event
	title = bronze_city_razing.20.title
	desc = bronze_city_razing.20.desc
	outcome = neutral

	illustration_tags = {
		10 = exterior
		10 = interior
	}

	trigger = {
		scope:bronze_ruins_site ?= { has_variable = bronze_ruins_expedition_underway }
	}

	option = {
		name = bronze_city_razing.20.a
		custom_tooltip = bronze_city_razing_20a_tt
		add_gold = -20
		scope:bronze_ruins_site = {
			set_variable = { name = be_ruin_expedition_cautious value = 1 }
			change_variable = { name = be_ruin_expedition_supplies add = 20 }
			change_variable = { name = be_ruin_expedition_morale add = 5 }
		}
		be_ruin_expedition_minor_progress_effect = yes
		be_ruin_expedition_schedule_next_story_event_effect = yes
	}

	option = {
		name = bronze_city_razing.20.b
		custom_tooltip = bronze_city_razing_20b_tt
		scope:bronze_ruins_site = {
			set_variable = { name = be_ruin_expedition_reckless value = 1 }
			change_variable = { name = be_ruin_expedition_supplies subtract = 10 }
			change_variable = { name = be_ruin_expedition_morale subtract = 5 }
		}
		be_ruin_expedition_major_progress_effect = yes
		be_ruin_expedition_raise_risk_effect = yes
		be_ruin_expedition_schedule_next_story_event_effect = yes
	}

	option = {
		name = bronze_city_razing.20.c
		custom_tooltip = bronze_city_razing_20c_tt
		add_gold = -15
		scope:bronze_ruins_site = {
			set_variable = { name = be_ruin_expedition_scholarly value = 1 }
			change_variable = { name = be_ruin_expedition_success_chance add = 5 }
			change_variable = { name = be_ruin_expedition_supplies add = 5 }
		}
		be_ruin_expedition_minor_progress_effect = yes
		be_ruin_expedition_schedule_next_story_event_effect = yes
	}
}

			has_variable = bronze_ruins_expedition_underway
			exists = owner
			NOT = { owner = root }
