bronze_volcanic_eruptions.1 = { # Mount Merapi - c.1180 BC ±75 years - VEI 4 - source javanese_mountains2 -> target borobudur (mapped)
	type = country_event
	title = bronze_volcanic_eruptions_minor.title
	desc = bronze_volcanic_eruptions_minor.desc
	outcome = negative
	fire_only_once = yes

	trigger = {
		is_real_country = yes
		owns = location:borobudur
		current_year >= 30
		current_year <= 30
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
		location:borobudur = {
			save_scope_as = target_location
			add_vfx = {
				name = volcano_eruption_test
				duration = 0
			}
		}
	}

	option = {
		name = volcano_events_minor.a
		historical_option = yes
		scope:target_location = { minor_volcano_damage = yes }
	}

	option = {
		name = volcano_events_minor.b
		scope:target_location = {
			minor_volcano_damage_cost = yes
			minor_volcano_damage_with_rebuild = yes
		}
	}
}
