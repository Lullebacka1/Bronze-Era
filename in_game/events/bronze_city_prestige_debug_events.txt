namespace = bronze_city_prestige_debug

# Console-only test menu. This event is never registered in an on-action.
# Command: event bronze_city_prestige_debug.1
bronze_city_prestige_debug.1 = {
	type = country_event
	title = bronze_city_prestige_debug.1.title
	desc = bronze_city_prestige_debug.1.desc
	outcome = neutral
	option = {
		name = bronze_city_prestige_debug.1.a
		if = {
			limit = { has_global_variable = bronze_city_prestige_manager }
			global_var:bronze_city_prestige_manager ?= {
				bronze_city_prestige_update_world_ranking = yes
				bronze_city_prestige_rebuild_ledger_list = yes
			}
		}
	}
	option = {
		name = bronze_city_prestige_debug.1.b
		remove_variable = bronze_city_mission_active
		remove_variable = bronze_city_mission_window
		set_variable = { name = bronze_city_mission_window value = 1 years = 5 }
	}
	option = {
		name = bronze_city_prestige_debug.1.c
		capital ?= {
			set_variable = { name = bronze_city_decline_level value = 1 }
			bronze_city_prestige_update_decline = yes
			bronze_city_prestige_commit_score_change = yes
		}
	}
	option = {
		name = bronze_city_prestige_debug.1.d
		capital ?= {
			set_variable = { name = bronze_city_decline_level value = 2 }
			bronze_city_prestige_update_decline = yes
			bronze_city_prestige_commit_score_change = yes
		}
	}
	option = {
		name = bronze_city_prestige_debug.1.e
		capital ?= {
			set_variable = { name = bronze_city_decline_level value = 3 }
			bronze_city_prestige_update_decline = yes
			bronze_city_prestige_commit_score_change = yes
		}
	}
	option = {
		name = bronze_city_prestige_debug.1.f
		capital ?= {
			set_variable = { name = bronze_city_decline_level value = 4 }
			set_variable = { name = bronze_city_decline_stage_4_years value = 20 }
			save_scope_as = bronze_declining_city
			root = {
				set_variable = {
					name = bronze_city_decline_event_city
					value = scope:bronze_declining_city
				}
			}
		}
		trigger_event_non_silently = {
			id = bronze_city_development.330
			days = 1
		}
	}
	option = {
		name = bronze_city_prestige_debug.1.g
		capital ?= {
			set_variable = { name = bronze_city_decline_level value = 0 }
			set_variable = { name = bronze_city_decline_bad_cycles value = 0 }
			set_variable = { name = bronze_city_decline_good_cycles value = 0 }
			remove_variable = bronze_city_decline_stage_4_years
			remove_location_modifier = bronze_city_decline_stage_1
			remove_location_modifier = bronze_city_decline_stage_2
			remove_location_modifier = bronze_city_decline_stage_3
			remove_location_modifier = bronze_city_decline_stage_4
			set_variable = { name = bronze_city_decline_prestige_penalty value = 0 }
			bronze_city_prestige_commit_score_change = yes
		}
	}
}

# Command: event bronze_city_prestige_debug.2
# Advances the currently stored pyramid project to its next decision event.
bronze_city_prestige_debug.2 = {
	type = country_event
	title = bronze_city_prestige_debug.2.title
	desc = bronze_city_prestige_debug.2.desc
	outcome = neutral
	option = {
		name = bronze_city_prestige_debug.2.a
		if = {
			limit = { var:bronze_great_pyramid_project_phase <= 1 }
			trigger_event_non_silently = { id = bronze_city_monuments.100 days = 1 }
		}
		else_if = {
			limit = { var:bronze_great_pyramid_project_phase = 2 }
			trigger_event_non_silently = { id = bronze_city_monuments.101 days = 1 }
		}
		else_if = {
			limit = { var:bronze_great_pyramid_project_phase = 3 }
			trigger_event_non_silently = { id = bronze_city_monuments.102 days = 1 }
		}
		else_if = {
			limit = { var:bronze_great_pyramid_project_phase = 4 }
			trigger_event_non_silently = { id = bronze_city_monuments.103 days = 1 }
		}
		else_if = {
			limit = { var:bronze_great_pyramid_project_phase = 5 }
			trigger_event_non_silently = { id = bronze_city_monuments.104 days = 1 }
		}
		else = {
			trigger_event_non_silently = { id = bronze_city_monuments.105 days = 1 }
		}
	}
}

# Command: event bronze_city_prestige_debug.3
# Opens the normal resolution event for the currently active urban mission.
# The regular event still decides whether the success condition is met.
bronze_city_prestige_debug.3 = {
	type = country_event
	title = bronze_city_prestige_debug.3.title
	desc = bronze_city_prestige_debug.3.desc
	outcome = neutral
	option = {
		name = bronze_city_prestige_debug.3.a
		if = {
			limit = { var:bronze_city_mission_type = 1 }
			trigger_event_non_silently = { id = bronze_city_development.200 days = 1 }
		}
		else_if = {
			limit = { var:bronze_city_mission_type = 2 }
			trigger_event_non_silently = { id = bronze_city_development.201 days = 1 }
		}
		else_if = {
			limit = { var:bronze_city_mission_type = 3 }
			trigger_event_non_silently = { id = bronze_city_development.202 days = 1 }
		}
		else_if = {
			limit = { var:bronze_city_mission_type = 4 }
			trigger_event_non_silently = { id = bronze_city_development.203 days = 1 }
		}
		else_if = {
			limit = { var:bronze_city_mission_type = 5 }
			trigger_event_non_silently = { id = bronze_city_development.204 days = 1 }
		}
	}
}

# Command: event bronze_city_prestige_debug.4
# Deterministically creates a test rivalry when two eligible cities exist.
# A neighboring foreign city is preferred; a second domestic city is fallback.
bronze_city_prestige_debug.4 = {
	type = country_event
	title = bronze_city_prestige_debug.4.title
	desc = bronze_city_prestige_debug.4.desc
	outcome = neutral
	option = {
		name = bronze_city_prestige_debug.4.a
		remove_variable = bronze_city_rivalry_country_cooldown
		random_owned_location = {
			limit = { bronze_city_prestige_is_great_or_higher = yes }
			save_scope_as = bronze_city_rivalry_home
		}
		random_neighbor_country = {
			limit = {
				is_real_country = yes
				any_owned_location = {
					bronze_city_prestige_is_great_or_higher = yes
				}
			}
			save_scope_as = bronze_city_rivalry_foreign_owner
			random_owned_location = {
				limit = { bronze_city_prestige_is_great_or_higher = yes }
				save_scope_as = bronze_city_rivalry_other
			}
		}
		if = {
			limit = {
				exists = scope:bronze_city_rivalry_home
				exists = scope:bronze_city_rivalry_other
			}
			set_variable = {
				name = bronze_city_rivalry_home_city
				value = scope:bronze_city_rivalry_home
			}
			set_variable = {
				name = bronze_city_rivalry_other_city
				value = scope:bronze_city_rivalry_other
			}
			set_variable = {
				name = bronze_city_rivalry_other_owner
				value = scope:bronze_city_rivalry_foreign_owner
			}
			trigger_event_non_silently = {
				id = bronze_city_development.320
				days = 1
			}
		}
		else_if = {
			limit = { exists = scope:bronze_city_rivalry_home }
			random_owned_location = {
				limit = {
					bronze_city_prestige_is_great_or_higher = yes
					NOT = { this = scope:bronze_city_rivalry_home }
				}
				save_scope_as = bronze_city_rivalry_other
			}
			if = {
				limit = { exists = scope:bronze_city_rivalry_other }
				set_variable = {
					name = bronze_city_rivalry_home_city
					value = scope:bronze_city_rivalry_home
				}
				set_variable = {
					name = bronze_city_rivalry_other_city
					value = scope:bronze_city_rivalry_other
				}
				trigger_event_non_silently = {
					id = bronze_city_development.321
					days = 1
				}
			}
		}
	}
}
