bronze_city_razing.10 = {
	type = country_event
	title = bronze_city_razing.10.title
	desc = bronze_city_razing.10.desc
	outcome = neutral

	illustration_tags = {
		10 = exterior
	}

	trigger = {
		OR = {
			scope:bronze_ruins_site ?= { has_variable = bronze_ruins_expedition_underway }
			any_owned_location = {
				has_variable = bronze_ruins_expedition_underway
			}
		}
	}

	immediate = {
		if = {
			limit = { NOT = { scope:bronze_ruins_site ?= { has_variable = bronze_ruins_expedition_underway } } }
			random_owned_location = {
				limit = { has_variable = bronze_ruins_expedition_underway }
				save_scope_as = bronze_ruins_site
			}
		}
	}

	option = {
		name = bronze_city_razing.10.a
		custom_tooltip = bronze_city_razing_10a_tt
		add_gold = -20
		if = {
			limit = { scope:bronze_ruins_site ?= { var:be_ruin_expedition_success_chance >= 75 } }
			random_list = {
				10 = { trigger_event_non_silently = { id = bronze_city_razing.11 days = 90 } }
				30 = { trigger_event_non_silently = { id = bronze_city_razing.12 days = 90 } }
				30 = { trigger_event_non_silently = { id = bronze_city_razing.13 days = 90 } }
				25 = { trigger_event_non_silently = { id = bronze_city_razing.14 days = 90 } }
				5 = { trigger_event_non_silently = { id = bronze_city_razing.15 days = 90 } }
			}
		}
		else_if = {
			limit = { scope:bronze_ruins_site ?= { var:be_ruin_expedition_success_chance >= 60 } }
			random_list = {
				15 = { trigger_event_non_silently = { id = bronze_city_razing.11 days = 90 } }
				25 = { trigger_event_non_silently = { id = bronze_city_razing.12 days = 90 } }
				25 = { trigger_event_non_silently = { id = bronze_city_razing.13 days = 90 } }
				20 = { trigger_event_non_silently = { id = bronze_city_razing.14 days = 90 } }
				15 = { trigger_event_non_silently = { id = bronze_city_razing.15 days = 90 } }
			}
		}
		else_if = {
			limit = { scope:bronze_ruins_site ?= { var:be_ruin_expedition_success_chance >= 45 } }
			random_list = {
				25 = { trigger_event_non_silently = { id = bronze_city_razing.11 days = 90 } }
				20 = { trigger_event_non_silently = { id = bronze_city_razing.12 days = 90 } }
				20 = { trigger_event_non_silently = { id = bronze_city_razing.13 days = 90 } }
				15 = { trigger_event_non_silently = { id = bronze_city_razing.14 days = 90 } }
				20 = { trigger_event_non_silently = { id = bronze_city_razing.15 days = 90 } }
			}
		}
		else = {
			random_list = {
				30 = { trigger_event_non_silently = { id = bronze_city_razing.11 days = 90 } }
				15 = { trigger_event_non_silently = { id = bronze_city_razing.12 days = 90 } }
				15 = { trigger_event_non_silently = { id = bronze_city_razing.13 days = 90 } }
				10 = { trigger_event_non_silently = { id = bronze_city_razing.14 days = 90 } }
				30 = { trigger_event_non_silently = { id = bronze_city_razing.15 days = 90 } }
			}
		}
	}

	option = {
		name = bronze_city_razing.10.b
		custom_tooltip = bronze_city_razing_10b_tt
		bronze_ruins_finish_expedition = yes
		add_prestige = prestige_weak_penalty
	}
}
