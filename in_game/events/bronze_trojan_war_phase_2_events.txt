join_trojan_War.1 = {
    type = country_event
    title = join_trojan_War.t
    desc = join_trojan_War.d
    is_triggered_only = yes
    hidden = no 
    
immediate = {
    save_scope_as = actor
    random_current_war = {
        limit = {
            attacker_leader = c:0001G
            defender_leader = c:WILUS
        }
        save_scope_as = trojan_war
    }
}

option = {
    name = "join_trojan_war"
    if = {
        limit = { exists = scope:trojan_war }
        join_war_as_defender = { war = scope:trojan_war }
    }
}
}
