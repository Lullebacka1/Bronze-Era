namespace = bronze_city_prestige

# ============================================================================
# PRESTIGIOUS CITY EVENTS
# ============================================================================
# These events are sampled by bronze_city_prestige_urban_event_pulse. Their
# country cooldown prevents repeated urban pop-ups, while location modifiers
# feed directly back into the next City Prestige ranking.
# ============================================================================

bronze_city_prestige.1 = {
	type = country_event
	title = bronze_city_prestige.1.title
	desc = bronze_city_prestige.1.desc
	outcome = positive

	illustration_tags = {
		10 = interior
	}

	trigger = {
		NOT = { has_variable = bronze_city_prestige_event_cooldown }
		any_owned_location = {
			bronze_city_prestige_scholars_event_location = yes
		}
	}

	immediate = {
		set_variable = {
			name = bronze_city_prestige_event_cooldown
			value = 1
			years = 3
		}
		random_owned_location = {
			limit = { bronze_city_prestige_scholars_event_location = yes }
			save_scope_as = bronze_city
		}
	}

	option = {
		name = bronze_city_prestige.1.a
		add_gold = -25
		add_prestige = prestige_weak_bonus
		add_country_modifier = {
			modifier = bronze_city_scholarly_patronage
			years = 6
			mode = add_and_extend
		}
		ruler_or_regent ?= {
			add_character_modifier = {
				modifier = bronze_city_scholarly_court
				years = 6
				mode = add_and_extend
			}
		}
		scope:bronze_city ?= {
			add_location_modifier = {
				modifier = bronze_city_scholars_gathered
				years = 8
				mode = add_and_extend
			}
			change_prosperity = prosperity_weak_bonus
		}
	}

	option = {
		name = bronze_city_prestige.1.b
		add_prestige = prestige_weak_bonus
		scope:bronze_city ?= {
			add_location_modifier = {
				modifier = bronze_city_scholars_gathered
				years = 3
				mode = add_and_extend
			}
		}
	}
}

bronze_city_prestige.5 = {
	type = country_event
	title = bronze_city_prestige.5.title
	desc = bronze_city_prestige.5.desc
	outcome = negative

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	trigger = {
		NOT = { has_variable = bronze_city_prestige_event_cooldown }
		any_owned_location = {
			bronze_city_prestige_famine_event_location = yes
		}
	}

	immediate = {
		set_variable = {
			name = bronze_city_prestige_event_cooldown
			value = 1
			years = 3
		}
		random_owned_location = {
			limit = { bronze_city_prestige_famine_event_location = yes }
			save_scope_as = bronze_city
		}
	}

	option = {
		name = bronze_city_prestige.5.a
		add_gold = -50
		add_prestige = prestige_weak_penalty
		scope:bronze_city ?= {
			add_location_modifier = {
				modifier = bronze_city_famine
				years = 3
				mode = add_and_extend
			}
			province = {
				change_province_food_percentage = 0.20
			}
			every_pop = {
				limit = { pop_size > 0.02 }
				add_pop_size = {
					value = pop_size
					multiply = -0.01
				}
			}
		}
	}

	option = {
		name = bronze_city_prestige.5.b
		add_stability = stability_weak_penalty
		add_prestige = prestige_mild_penalty
		scope:bronze_city ?= {
			add_location_modifier = {
				modifier = bronze_city_famine
				years = 6
				mode = add_and_extend
			}
			province = {
				change_province_food_percentage = 0.08
			}
			change_prosperity = prosperity_mild_penalty
			every_pop = {
				limit = { pop_size > 0.02 }
				add_pop_size = {
					value = pop_size
					multiply = -0.03
				}
			}
		}
	}
}

bronze_city_prestige.7 = {
	type = country_event
	title = bronze_city_prestige.7.title
	desc = bronze_city_prestige.7.desc
	outcome = negative

	illustration_tags = {
		10 = exterior
	}

	trigger = {
		NOT = { has_variable = bronze_city_prestige_event_cooldown }
		any_owned_location = {
			bronze_city_prestige_exodus_event_location = yes
		}
	}

	immediate = {
		set_variable = {
			name = bronze_city_prestige_event_cooldown
			value = 1
			years = 3
		}
		random_owned_location = {
			limit = { bronze_city_prestige_exodus_event_location = yes }
			save_scope_as = bronze_city
		}
		random_owned_location = {
			limit = {
				NOT = { this = scope:bronze_city }
			}
			save_scope_as = bronze_city_refuge
		}
	}

	option = {
		name = bronze_city_prestige.7.a
		add_gold = -30
		add_prestige = prestige_weak_penalty
		scope:bronze_city ?= {
			add_location_modifier = {
				modifier = bronze_city_population_exodus
				years = 4
				mode = add_and_extend
			}
			every_pop = {
				limit = { pop_size > 0.02 }
				add_pop_size = {
					value = pop_size
					multiply = -0.01
				}
			}
		}
		if = {
			limit = {
				exists = scope:bronze_city
				exists = scope:bronze_city_refuge
			}
			add_migration = {
				owner = root
				from = scope:bronze_city.province_definition
				to = scope:bronze_city_refuge.province_definition
				type = pop_type:peasants
				amount = 0.01
				months = 24
			}
		}
	}

	option = {
		name = bronze_city_prestige.7.b
		add_stability = stability_weak_penalty
		add_prestige = prestige_mild_penalty
		scope:bronze_city ?= {
			add_location_modifier = {
				modifier = bronze_city_population_exodus
				years = 8
				mode = add_and_extend
			}
			change_prosperity = prosperity_severe_penalty
			every_pop = {
				limit = { pop_size > 0.02 }
				add_pop_size = {
					value = pop_size
					multiply = -0.04
				}
			}
		}
	}
}
