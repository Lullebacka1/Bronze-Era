namespace = levant_silver_debasement

scripted_effect update_silver_market = {
	every_market_present_in_country = {

		remove_temporary_demand = demand:levantine_silver_crisis_debasement_start
		remove_temporary_demand = demand:levantine_silver_crisis

		add_temporary_demand = {
			type = demand:levantine_silver_crisis_debasement_start
			scale = global_var:levant_silver_crisis_copper_demand
			years = 100
		}

		add_temporary_demand = {
			type = demand:levantine_silver_crisis
			scale = global_var:levant_silver_crisis_silver_demand
			years = 100
		}
	}
}
