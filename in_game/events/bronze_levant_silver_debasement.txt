levant_silver_debasement.1 = {
	type = country_event
	title = levant_silver_debasement.1.title
	desc = levant_silver_debasement.1.desc
	category = situation_event
	image = "gfx/interface/illustrations/situation/levant_silver_debasement_1.dds"
	outcome = neutral

	trigger = {
		NOT = { has_variable = levant_silver_debasement_intro_seen }
		has_ports = yes
		levant_silver_debasement_participant = yes
		current_date < 5.1.1
		current_date > 3.1.1
	}

	immediate = {
		set_variable = { name = levant_silver_debasement_intro_seen value = 1 years = 50 }
	}

	option = {
		name = levant_silver_debasement.1.a
		custom_tooltip = levant_silver_debasement.1.a_tt

		add_stability = stability_weak_bonus
		add_country_modifier = {
			modifier = levant_silver_debasement_initial_hit
			years = 10
		}
	}

	option = {
		name = levant_silver_debasement.1.b
		custom_tooltip = levant_silver_debasement.1.b_tt
		
		add_diplomats = diplomats_weak_bonus
		add_country_modifier = {
			modifier = levant_silver_debasement_initial_hit
			years = 10
		}
	}

	option = {
		name = levant_silver_debasement.1.c
		custom_tooltip = levant_silver_debasement.1.c_tt

		add_devotion = devotion_weak_bonus
		add_country_modifier = {
			modifier = levant_silver_debasement_initial_hit
			years = 10
		}
	}
}
