namespace = bronze_city_development

# ============================================================================
# URBAN PRESTIGE MISSIONS
# ============================================================================
# Every event retrieves the exact selected city from the country's persistent
# object variable. Success is available only if the objective still holds.
# Abandonment always has a real prestige, prosperity, and cooldown consequence.
# ============================================================================

bronze_city_development.200 = {
	type = country_event
	title = bronze_city_development.200.title
	desc = bronze_city_development.200.desc
	outcome = neutral
	immediate = {
		var:bronze_city_mission_city ?= { save_scope_as = bronze_city_mission_site }
	}
	option = {
		name = bronze_city_development.200.a
		trigger = {
			scope:bronze_city_mission_site ?= {
				owner = root
				bronze_city_prestige_is_urban = yes
				prosperity >= 0
			}
		}
		add_prestige = prestige_mild_bonus
		scope:bronze_city_mission_site = {
			if = {
				limit = { NOT = { has_variable = bronze_city_mission_prestige } }
				set_variable = { name = bronze_city_mission_prestige value = 0 }
			}
			change_variable = { name = bronze_city_mission_prestige add = 15 }
			set_variable = { name = bronze_city_mission_cooldown value = 1 years = 15 }
			remove_location_modifier = bronze_city_mission_underway
			add_location_modifier = {
				modifier = bronze_city_grand_festival_legacy
				years = 8
				mode = add_and_extend
			}
			bronze_city_prestige_reduce_decline = yes
			bronze_city_prestige_commit_score_change = yes
		}
		remove_variable = bronze_city_mission_active
		remove_variable = bronze_city_mission_type
		remove_variable = bronze_city_mission_city
	}
	option = {
		name = bronze_city_development.mission_failed
		add_prestige = prestige_weak_penalty
		scope:bronze_city_mission_site ?= {
			set_variable = { name = bronze_city_mission_cooldown value = 1 years = 8 }
			remove_location_modifier = bronze_city_mission_underway
			change_prosperity = prosperity_weak_penalty
			bronze_city_prestige_commit_score_change = yes
		}
		remove_variable = bronze_city_mission_active
		remove_variable = bronze_city_mission_type
		remove_variable = bronze_city_mission_city
	}
}

bronze_city_development.330 = {
	type = country_event
	title = bronze_city_development.330.title
	desc = bronze_city_development.330.desc
	outcome = negative
	immediate = {
		var:bronze_city_decline_event_city ?= {
			save_scope_as = bronze_declining_city
		}
	}
	option = {
		name = bronze_city_development.330.a
		trigger = {
			scope:bronze_declining_city ?= {
				owner = root
				bronze_city_prestige_is_urban = yes
			}
			gold >= 200
			manpower = { value >= 100 }
		}
		add_gold = -200
		add_manpower_pool = -100
		add_prestige = prestige_mild_bonus
		scope:bronze_declining_city = {
			set_variable = { name = bronze_city_decline_level value = 1 }
			set_variable = { name = bronze_city_decline_bad_cycles value = 0 }
			set_variable = { name = bronze_city_decline_good_cycles value = 0 }
			remove_variable = bronze_city_decline_stage_4_years
			remove_location_modifier = bronze_city_decline_stage_2
			remove_location_modifier = bronze_city_decline_stage_3
			remove_location_modifier = bronze_city_decline_stage_4
			add_location_modifier = {
				modifier = bronze_city_decline_stage_1
				years = -1
				mode = replace
			}
			set_variable = { name = bronze_city_decline_prestige_penalty value = 10 }
			change_prosperity = prosperity_mild_bonus
			province = { change_province_food_percentage = 0.15 }
			bronze_city_prestige_commit_score_change = yes
		}
		remove_variable = bronze_city_decline_event_city
	}
	option = {
		name = bronze_city_development.330.b
		add_gold = -50
		add_stability = stability_weak_penalty
		scope:bronze_declining_city ?= {
			change_prosperity = prosperity_severe_penalty
			set_variable = { name = bronze_city_decline_stage_4_years value = 0 }
			every_pop = {
				limit = { pop_size > 0.02 }
				add_pop_size = {
					value = pop_size
					multiply = -0.05
				}
			}
			bronze_city_prestige_commit_score_change = yes
		}
		remove_variable = bronze_city_decline_event_city
	}
	option = {
		name = bronze_city_development.330.c
		add_prestige = prestige_severe_penalty
		scope:bronze_declining_city ?= {
			bronze_historic_monuments_snapshot_for_ruins = yes
			if = {
				limit = { NOT = { has_variable = be_ruins_stats_generated } }
				be_ruins_generate_stats_effect = yes
			}
			every_buildings_in_location = {
				save_temporary_scope_as = bronze_city_target_building
				PREV = {
					destroy_building = scope:bronze_city_target_building
				}
			}
			bronze_city_create_ancient_ruins_effect = yes
		}
		if = {
			limit = { has_global_variable = bronze_city_prestige_manager }
			global_var:bronze_city_prestige_manager ?= {
				bronze_city_prestige_update_world_ranking = yes
				bronze_city_prestige_rebuild_ledger_list = yes
			}
		}
		remove_variable = bronze_city_decline_event_city
	}
}
